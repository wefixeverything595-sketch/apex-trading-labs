# APEX COPY ENGINE - ASYNC REDIS REPLICATION WORKER
import asyncio
import json
import redis.asyncio as redis
import MetaTrader5 as mt5

r = redis.Redis(host='localhost', port=6379, db=0)

async def process_signal(trade_event):
    symbol = trade_event['symbol']
    action = trade_event['type'] # BUY or SELL
    lots = trade_event['lots']
    
    # Calculate follower lot size based on relative equity
    follower_equity = 10000.0
    master_equity = 50000.0
    scaled_lots = round((follower_equity / master_equity) * lots, 2)

    # MT5 Order Dispatch
    request = {
        "action": mt5.TRADE_ACTION_DEAL,
        "symbol": symbol,
        "volume": scaled_lots,
        "type": mt5.ORDER_TYPE_BUY if action == 'BUY' else mt5.ORDER_TYPE_SELL,
        "price": mt5.symbol_info_tick(symbol).ask if action == 'BUY' else mt5.symbol_info_tick(symbol).bid,
        "deviation": 20,
        "magic": 888111,
        "comment": "ApexCopyTrade",
    }
    result = mt5.order_send(request)
    print(f"Replicated Trade Code: {result.retcode}")

async def main():
    pubsub = r.pubsub()
    await pubsub.subscribe('master_signals')
    async for message in pubsub.listen():
        if message['type'] == 'message':
            data = json.loads(message['data'].decode('utf-8'))
            await process_signal(data)

if __name__ == '__main__':
    asyncio.run(main())

   if(trans.type == TRADE_TRANSACTION_ORDER_ADD)
   {
      CJAVal json;
      json["symbol"] = trans.symbol;
      json["type"] = (trans.order_type == ORDER_TYPE_BUY) ? "BUY" : "SELL";
      json["lots"] = trans.volume;
      
      string payload = json.Serialize();
      
      // Send payload via WebRequest to FastAPI/Redis Bridge
      char data[], result_data[];
      string headers = "Content-Type: application/json\r\n";
      StringToCharArray(payload, data, 0, WHOLE_ARRAY, CP_UTF8);
      
      int res = WebRequest("POST", "http://localhost:8000/api/signal", 
                          headers, 1000, data, result_data, headers);

   }
